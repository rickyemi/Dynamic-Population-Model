def summarise_tipping_points(result: dict) -> pd.DataFrame:
    """The rebuilt tipping points, the headline output of the tab."""
    tp_start = result["tp_title_row"] + 2
    rows = []
    for (r, c), v in sorted(result["computed"].items()):
        if c == 13 and r >= tp_start:
            cells = {cc: vv for (rr, cc, vv, _) in result["cells"] if rr == r for cc, vv
                     in [(cc, vv)]}
            rows.append({"Sex": cells.get(10), "ERR A/ERR B": cells.get(11),
                         "Gateway": cells.get(12),
                         "In combination with cigarettes": f"{v:.1%}",
                         "Exclusively": f"{result['computed'][(r, 14)]:.1%}"})
    return pd.DataFrame(rows)


def validate_workbook(result: dict) -> None:
    """Structural checks on the written workbook, independent of any source tab.

    The formula count is fixed by the geometry: 11 per interpolation row, 4 per no quitting table
    and 2 per tipping point row. Checking it catches a layout change that still produces a
    plausible looking sheet.
    """
    wb = load_workbook(result["output_file"])
    expected_tabs = [result["loaded"]["sheet_names"]["detailed"][:31],
                     result["loaded"]["sheet_names"]["master"][:31], OUTPUT_DIFF_SHEET]
    assert wb.sheetnames == expected_tabs, f"Tabs are {wb.sheetnames}, expected {expected_tabs}"

    ws = wb[OUTPUT_DIFF_SHEET]
    n_gateways = len(result["loaded"]["gateways"])
    for bi in range(n_gateways):
        pos = block_rows(1 + bi * BLOCK_STRIDE)
        assert ws.cell(row=pos["left_header"], column=1).value == "Model Name"
        assert ws.cell(row=pos["left_header"], column=5).value in GATEWAY_FRACTION.values()
        for sex in SEX_ORDER:
            hdr = pos["dual_header"][sex]
            assert [ws.cell(row=hdr, column=COL["H"] + i).value
                    for i in range(11)] == DUAL_USE_STEPS
        assert ws.cell(row=pos["nq_header"], column=3).value == "No quitting"

    n_formulas = sum(1 for _, _, v, _ in result["cells"]
                     if isinstance(v, str) and v.startswith("="))
    expected = n_gateways * (4 * 2 * 11 + 4) + len(TP_SEXES) * n_gateways * len(MATCHED_PAIRS) * 2
    assert n_formulas == expected, f"{n_formulas} formulas written, expected {expected}"

    print(f"Workbook checks passed: {len(wb.sheetnames)} tabs, {n_gateways} gateway block(s), "
          f"{n_formulas} live formulas in {OUTPUT_DIFF_SHEET}.")


def run_optimizer(data_dir=None, out_dir=None, filename=None, validate: bool = True,
                  verbose: bool = True) -> dict:
    """Rebuild the Difference tab, write the workbook, and validate it."""
    result = run_pipeline(data_dir=data_dir, out_dir=out_dir, filename=filename)
    if verbose:
        print("\nRebuilt tipping points:")
        print(summarise_tipping_points(result).to_string(index=False))
    if validate:
        # Only compares when the input still carries its original Difference tab
        validate_against_source(result)
        validate_workbook(result)
    return result

# SELF TEST

#
# A synthetic workbook meeting all three input requirements, with means perturbed from the shape of
# a real export so the numbers differ while the structure does not. It carries no Difference tab,
# so `validate_against_source` correctly reports nothing to compare and only the structural checks
# run: that is the case a new product presents, and it has to work.

SELF_TEST_PRODUCT = "Product_S"
SELF_TEST_INITIALS = "YO"
SELF_TEST_FILE = f"Dual_Use_Calculations_TwoAge_B01_{SELF_TEST_PRODUCT}_September_4_2026.xlsx"

# (gateway, sex, ERR) -> mean, the shape a real export has
SELF_TEST_DETAILED = {
    ("G10", "Male", 0.05): 3065, ("G10", "Male", 0.10): 2777,
    ("G10", "Male", 1.05): -2231, ("G10", "Male", 1.10): -2465,
    ("G10", "Female", 0.05): 2603, ("G10", "Female", 0.10): 2368,
    ("G10", "Female", 1.05): -1968, ("G10", "Female", 1.10): -2186,
    ("G25", "Male", 0.05): 2601, ("G25", "Male", 0.10): 2342,
    ("G25", "Male", 1.05): -2209, ("G25", "Male", 1.10): -2424,
    ("G25", "Female", 0.05): 2189, ("G25", "Female", 0.10): 1979,
    ("G25", "Female", 1.05): -1947, ("G25", "Female", 1.10): -2146,
}
SELF_TEST_MASTER = {
    ("G10", "Male", 0.05): 2679, ("G10", "Male", 0.10): 2047,
    ("G10", "Female", 0.05): 2294, ("G10", "Female", 0.10): 1783,
    ("G25", "Male", 0.05): 2260, ("G25", "Male", 0.10): 1697,
    ("G25", "Female", 0.05): 1916, ("G25", "Female", 0.10): 1462,
}
SELF_TEST_MORTALITY = {"Male": "JAGS_Male_2000", "Female": "JAGS_Female_2000"}


def build_self_test_workbook(folder, seed: int = 20260904) -> Path:
    """Write a synthetic Dual Use workbook meeting Requirements 1, 2 and 3."""
    rng = np.random.default_rng(seed)
    folder = Path(folder)
    folder.mkdir(parents=True, exist_ok=True)
    group = f"{SELF_TEST_PRODUCT}_{SELF_TEST_INITIALS}"

    def row(name, sex, err, mean, with_band=False):
        rec = {"Model Group": group, "Model Name": name, "Mortality Model": SELF_TEST_MORTALITY[sex],
               "ERR": err, "ERR Variability": "Variable",
               "TP Variability": "Smoking initiation and cessation variable",
               "Age Range": "68 - 72", "Node": "DIFF_ALL", "Mean": mean}
        if with_band:
            rec["95% PI"] = int(round(mean * 0.86))
            rec[""] = int(round(mean * 1.15))
        rec.update({"Trans. Prob Stand. Dev.": 0.01, "ERR Standard Deviation": 0.01,
                    "QALE Mean": "N/A", "LE Mean": "N/A",
                    "Creation Date": pd.Timestamp("2026-09-04 08:15:00"),
                    "Web Version": "3.0.0", "R Version": "1.8"})
        return rec

    perturb = lambda v: int(round(v * rng.uniform(0.93, 1.08)))
    detailed = [row(f"Master Model {gw} {SELF_TEST_PRODUCT} Pouch Quitting_TwoAge_5a_6a_14b_15a_B01",
                    sex, err, perturb(mean))
                for (gw, sex, err), mean in SELF_TEST_DETAILED.items()]
    master = [row(f"Master Model {gw} {SELF_TEST_PRODUCT}_TwoAge_5a_6a_14b_15a_B01",
                  sex, err, perturb(mean), with_band=True)
              for (gw, sex, err), mean in SELF_TEST_MASTER.items()]

    path = folder / SELF_TEST_FILE
    with pd.ExcelWriter(path) as xl:
        for frame, sheet in ((pd.DataFrame(detailed), "Detailed Results"),
                             (pd.DataFrame(master), "MasterModels TP-ID")):
            cols = list(frame.columns)
            stacked = pd.concat(
                [pd.DataFrame([["Detailed Model Results"] + [None] * (len(cols) - 1)]),
                 pd.DataFrame([cols]), pd.DataFrame(frame.values)], ignore_index=True)
            stacked.to_excel(xl, sheet_name=sheet, index=False, header=False)
    return path


def self_test(folder=None) -> dict:
    """Build a synthetic workbook, rebuild the tab from it and assert the structure."""
    folder = Path(folder or Path.cwd() / "self_test_dual_use")
    build_self_test_workbook(folder)
    result = run_optimizer(data_dir=folder, out_dir=folder / "output", validate=False,
                           verbose=False)

    assert result["product"] == SELF_TEST_PRODUCT, f"product read as {result['product']!r}"
    assert result["initials"] == SELF_TEST_INITIALS
    validate_workbook(result)

    tps = summarise_tipping_points(result)
    assert len(tps) == len(TP_SEXES) * len(result["loaded"]["gateways"]) * len(MATCHED_PAIRS)
    print(f"Self test passed: product {result['product']}, "
          f"{len(tps)} tipping point row(s), output {result['output_file'].name}")
    return result
